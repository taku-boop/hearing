function doPost(e) { try { // 1. スプレッドシートを取得 var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getActiveSheet(); // 2. HTMLから送られてきたJSONデータを取得 var params; if (e.postData && e.postData.contents) { params = JSON.parse(e.postData.contents); } else { params = e.parameter; } // 3. ご指定の列順（タイムスタンプ ➔ 日付 ➔ 対応者 ➔ ランク ➔ 会社名...）に合わせて配列化（全24列） var rowData = [ params._sentAt || '', // 1. タイムスタンプ params.date || '', // 2. 日付（10月22日 / 10月23日） params.staff || '', // 3. 対応者（秋元 / 山田 / 今堀 / 坂） params.rank || '', // 4. ランク（A / B / C） params.company || '', // 5. 会社名 params.department || '', // 6. 部署 params.name || '', // 7. 氏名 params.summary || '', // 8. AIまとめ（内容まとめ） params.phone || '', // 9. 電話 params.email || '', // 10. メール params.exhibition || '', // 11. 展示会名（空欄） params.q1 || '', // 12. Q1_興味のあるサービス params.q1_other || '', // 13. Q1_その他 params.q2 || '', // 14. Q2_スケジュール params.q2_other || '', // 15. Q2_その他 params.q3 || '', // 16. Q3_ロット規模 params.q3_other || '', // 17. Q3_その他 params.q4 || '', // 18. Q4_使用材質 params.q4_other || '', // 19. Q4_その他 params.q5 || '', // 20. Q5_現在の課題 params.q5_other || '', // 21. Q5_その他 params.q6 || '', // 22. Q6_工場見学 params.q6_visit_time || '', // 23. Q6_希望時期 params.memo || '' // 24. 備考 ]; // 4. A列〜X列（24列分）の最下行へ固定書き込み var targetRow = sheet.getLastRow() + 1; sheet.getRange(targetRow, 1, 1, rowData.length).setValues([rowData]); // 5. 成功レスポンスを返す return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { console.error('エラー発生:', error.toString()); return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
